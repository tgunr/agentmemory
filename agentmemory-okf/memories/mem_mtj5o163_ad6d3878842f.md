```bash
#!/bin/bash

# Base source folder
BASE_DIR="/alpha/sources"

# Function to check if a directory is a git repo
is_git_repo() {
    git -C "$1" rev-parse > /dev/null 2>&1
}

# Function to process directories
process_dir() {
    for dir in "$1"/*; do
        if [ -d "$dir" ]; then
            if is_git_repo "$dir"; then
                # Create submodule for git repo
                echo "Adding submodule for $dir"
                git submodule add "$dir" "$BASE_DIR/$(basename "$dir")"
            else
                # Commit the directory into the base folder if it's not a git repo
                echo "Committing $dir into $BASE_DIR"
                cp -r "$dir" "$BASE_DIR"
            fi
            # Recursively process subdirectories
            process_dir "$dir"
        fi
    done
}

# Start processing from the base directory
process_dir "$BASE_DIR"

echo "Process completed."
```
